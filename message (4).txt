-- Dynamically find the Player's Tycoon
local function getMyTycoon()
    local tycoons = workspace:FindFirstChild("Tycoons")
    if not tycoons then return nil end

    for _, tycoon in ipairs(tycoons:GetChildren()) do
        local ownerValue = tycoon:FindFirstChild("Owner") or tycoon:FindFirstChild("Player")
        if ownerValue and (ownerValue.Value == LocalPlayer or ownerValue.Value == LocalPlayer.Name) then
            return tycoon
        end
    end

    local fallback = tycoons:FindFirstChild(LocalPlayer.Name) or tycoons:FindFirstChild("8")
    if fallback then return fallback end

    return tycoons:GetChildren()[1]
end

-- Extremely Strict Cash-Only Filter
local function isSafeToBuy(button)
    local nameLower = string.lower(button.Name)
    
    -- 1. If the name contains "R$" or "robux", it is absolutely a Robux button.
    if string.find(nameLower, "r$") or string.find(nameLower, "r %$") or string.find(nameLower, "robux") then
        return false
    end
    
    -- 2. Check if it actually contains a normal cash "$" sign.
    -- If it does NOT have a "$", it's either free or a premium/Robux button. We skip it to be safe.
    local hasCashSign = string.find(nameLower, "%$")
    if not hasCashSign then
        return false
    end
    
    -- 3. Deep-scan descendants for any hidden Robux values or Developer Product objects
    for _, child in ipairs(button:GetDescendants()) do
        local childName = string.lower(child.Name)
        if string.find(childName, "robux") or string.find(childName, "gamepass") or string.find(childName, "product") then
            return false
        end
        
        -- Check any visible text label inside the button for Robux indicators
        if child:IsA("TextLabel") then
            local textLower = string.lower(child.Text)
            if string.find(textLower, "r%$") or string.find(textLower, "robux") then
                return false
            end
        end
    end
    
    -- 4. Check attributes for developer product/gamepass markers
    local attributes = button:GetAttributes()
    if attributes["ProductId"] or attributes["GamepassId"] or attributes["Robux"] then
        return false
    end
    
    -- If it passed all tests and has a "$" sign, it's safe!
    return true
end

-- Function to physically step on a button
local function touchButton(button)
    local touchPart = button:FindFirstChild("Head") 
        or button:FindFirstChild("Pad") 
        or button:FindFirstChild("Button") 
        or button:FindFirstChildWhichIsA("BasePart") 
        or button

    local character = LocalPlayer.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")

    if rootPart and touchPart then
        firetouchinterest(rootPart, touchPart, 0)
        task.wait(0.02)
        firetouchinterest(rootPart, touchPart, 1)
    end
end

-- 1. Strict Fast Auto-Buy Loop
task.spawn(function()
    while true do
        if AutoBuyEnabled then
            local myTycoon = getMyTycoon()
            local buttonsFolder = myTycoon and myTycoon:FindFirstChild("Buttons")
            
            if buttonsFolder then
                local buttons = buttonsFolder:GetChildren()
                if #buttons > 0 then
                    for _, button in ipairs(buttons) do
                        if not AutoBuyEnabled then break end 
                        
                        -- Strictly verify the button is safe before stepping on it
                        if isSafeToBuy(button) then
                            touchButton(button)
                            task.wait(0.1) -- Fast-buy delay
                        end
                    end
                end
            end
        end
        task.wait(0.5)
    end
end)

-- 2. Auto-Collect Money Loop
task.spawn(function()
    while true do
        if AutoCollectEnabled and ReplicaSignal then
            ReplicaSignal:FireServer(4, "collectMoney", LocalPlayer)
        end
        task.wait(1) 
    end
end)

-- Smooth UI Drag Script
local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then update(input) end
end)
