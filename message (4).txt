-- Strictly binds ONLY to the tycoon matching your chosen Hero slot
local function getMyTycoon()
    local tycoons = workspace:FindFirstChild("Tycoons")
    if not tycoons then return nil end

    local myTycoon = tycoons:FindFirstChild(ChosenHero)
    if myTycoon then
        return myTycoon
    end

    for _, tycoon in ipairs(tycoons:GetChildren()) do
        local ownerValue = tycoon:FindFirstChild("Owner") or tycoon:FindFirstChild("Player")
        if ownerValue and (ownerValue.Value == LocalPlayer or ownerValue.Value == LocalPlayer.Name) then
            return tycoon
        end
    end

    return nil
end

-- Extremely Strict Cash-Only Filter (Keeps you safe from Robux items)
local function isSafeToBuy(button)
    local nameLower = string.lower(button.Name)
    if string.find(nameLower, "r$") or string.find(nameLower, "r %$") or string.find(nameLower, "robux") then
        return false
    end
    
    local hasCashSign = string.find(nameLower, "%$")
    if not hasCashSign then
        return false
    end
    
    for _, child in ipairs(button:GetDescendants()) do
        local childName = string.lower(child.Name)
        if string.find(childName, "robux") or string.find(childName, "gamepass") or string.find(childName, "product") then
            return false
        end
        if child:IsA("TextLabel") then
            local textLower = string.lower(child.Text)
            if string.find(textLower, "r%$") or string.find(textLower, "robux") then
                return false
            end
        end
    end
    
    local attributes = button:GetAttributes()
    if attributes["ProductId"] or attributes["GamepassId"] or attributes["Robux"] then
        return false
    end
    
    return true
end

-- Function to physically step on a button/pad safely
local function touchPartSafely(touchPart)
    local character = LocalPlayer.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")

    if rootPart and touchPart then
        firetouchinterest(rootPart, touchPart, 0)
        task.wait(0.02)
        firetouchinterest(rootPart, touchPart, 1)
    end
end

-- 1. Strict Fast Auto-Buy Loop (Only for your Hero's base)
task.spawn(function()
    while true do
        if AutoBuyEnabled then
            local myTycoon = getMyTycoon()
            local buttonsFolder = myTycoon and myTycoon:FindFirstChild("Buttons")
            
            if buttonsFolder then
                local buttons = buttonsFolder:GetChildren()
                if #buttons > 0 then
                    for _, button in ipairs(buttons) do
                        if not AutoBuyEnabled then break end 
                        if isSafeToBuy(button) then
                            local touchPart = button:FindFirstChild("Head") 
                                or button:FindFirstChild("Pad") 
                                or button:FindFirstChild("Button") 
                                or button:FindFirstChildWhichIsA("BasePart") 
                                or button
                            touchPartSafely(touchPart)
                            task.wait(0.1)
                        end
                    end
                end
            end
        end
        task.wait(0.5)
    end
end)

-- 2. Brute-Force Map-Wide Auto-Collect Loop
task.spawn(function()
    while true do
        if AutoCollectEnabled then
            local tycoons = workspace:FindFirstChild("Tycoons")
            if tycoons then
                -- Loop through every single tycoon on the map
                for _, tycoon in ipairs(tycoons:GetChildren()) do
                    if not AutoCollectEnabled then break end
                    
                    -- Find collection pad descendants inside this tycoon
                    for _, descendant in ipairs(tycoon:GetDescendants()) do
                        if descendant:IsA("BasePart") and (descendant.Name == "Pad" or descendant.Name == "Collect" or descendant.Name == "Collector") then
                            local isCollector = descendant.Parent.Name == "Collector" or descendant.Parent.Name == "EssentialParts" or string.find(string.lower(descendant.Name), "collect")
                            
                            if isCollector then
                                -- 1. Read the current cash display on the pad to get the exact amount
                                local amountToCollect = 40 -- Default fallback if we can't find a label
                                local textLabel = descendant.Parent:FindFirstChildWhichIsA("TextLabel", true)
                                if textLabel then
                                    -- Strips letters, spaces, dollar signs, and commas to grab just the numbers
                                    local parsedValue = string.gsub(textLabel.Text, "[^%d]", "")
                                    if tonumber(parsedValue) and tonumber(parsedValue) > 0 then
                                        amountToCollect = tonumber(parsedValue)
                                    end
                                end

                                -- 2. Touch the pad (Only yours will actually succeed/give you money)
                                touchPartSafely(descendant)

                                -- 3. Run Cobalt Visual Money Update
                                if firesignal and ReplicaWrite then
                                    firesignal(ReplicaWrite.OnClientEvent, 4, 3, "Money", amountToCollect)
                                end
                            end
                        end
                    end
                end
            end
        end
        task.wait(0.5) -- Scans all pads on the map twice every second
    end
end)

-- 3. Auto-Clicker Loop (Only clicks inside your Hero's tycoon)
task.spawn(function()
    while true do
        if AutoClickEnabled then
            local myTycoon = getMyTycoon()
            if myTycoon then
                local clicker = myTycoon:FindFirstChild("clicker", true)
                if clicker then
                    local clickDetector = clicker:FindFirstChildOfClass("ClickDetector")
                    if clickDetector and fireclickdetector then
                        fireclickdetector(clickDetector)
                    end
                end
            end
        end
        task.wait(0.05)
    end
end)
