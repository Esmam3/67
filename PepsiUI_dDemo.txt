local iconsFetched = false
local function iconId(name)
	if name == nil or name == "" then
		return nil
	end
	if type(name) == "number" then
		return "rbxassetid://" .. name
	end
	name = tostring(name)
	if name:find("rbxassetid://", 1, true) or name:find("http", 1, true) then
		return name
	end
	if tonumber(name) then
		return "rbxassetid://" .. name
	end
	name = name:gsub("^lucide:", ""):lower()
	if Icons[name] then
		return "rbxassetid://" .. Icons[name]
	end
	if not iconsFetched then
		iconsFetched = true
		pcall(function()
			local full = loadstring(game:HttpGet("https://raw.githubusercontent.com/Footagesus/Icons/refs/heads/main/lucide/dist/Icons.lua"))()
			for k, v in pairs(full) do
				local id = tostring(v):match("%d+")
				if id and not Icons[k] then
					Icons[k] = tonumber(id)
				end
			end
		end)
		if Icons[name] then
			return "rbxassetid://" .. Icons[name]
		end
	end
	return nil
end
Library.Icon = iconId

local function create(className, props, children)
	local object = Instance.new(className)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then
			object[k] = v
		end
	end
	for _, child in ipairs(children or {}) do
		child.Parent = object
	end
	if props and props.Parent then
		object.Parent = props.Parent
	end
	return object
end

local function tween(object, props, time, style, direction)
	local t = TweenService:Create(object, TweenInfo.new(time or 0.22, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

local function corner(radius, parent)
	return create("UICorner", { CornerRadius = typeof(radius) == "UDim" and radius or UDim.new(0, radius or 8), Parent = parent })
end

-- Accent gradients repaint when the theme accent changes.
local accentGradients = setmetatable({}, { __mode = "k" })
local accentSolids = setmetatable({}, { __mode = "k" })
local function accentGradient(parent, rotation)
	local g = create("UIGradient", {
		Color = ColorSequence.new(Theme.AccentA, Theme.AccentB),
		Rotation = rotation or 0,
		Parent = parent,
	})
	accentGradients[g] = true
	return g
end
local function accentSolid(object, property)
	accentSolids[object] = property
	object[property] = Theme.AccentA
end

function Library:SetAccent(a, b)
	Theme.AccentA, Theme.AccentB = a, b or a
	for g in pairs(accentGradients) do
		if g.Parent then
			g.Color = ColorSequence.new(Theme.AccentA, Theme.AccentB)
		end
	end
	for object, property in pairs(accentSolids) do
		if object.Parent then
			object[property] = Theme.AccentA
		end
	end
end

-- Ripple from where you pressed.
local function ripple(holder, input)
	local size = math.max(holder.AbsoluteSize.X, holder.AbsoluteSize.Y) * 1.6
	local x, y = holder.AbsoluteSize.X / 2, holder.AbsoluteSize.Y / 2
	if input and input.Position then
		x = input.Position.X - holder.AbsolutePosition.X
		y = input.Position.Y - holder.AbsolutePosition.Y
	end
	local circle = create("Frame", {
		Name = "Ripple",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(x, y),
		Size = UDim2.fromOffset(0, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = 0.82,
		ZIndex = holder.ZIndex + 5,
		Parent = holder,
	})
	corner(UDim.new(1, 0), circle)
	tween(circle, { Size = UDim2.fromOffset(size, size), BackgroundTransparency = 1 }, 0.55, Enum.EasingStyle.Quad)
	task.delay(0.6, function()
		circle:Destroy()
	end)
end

-- Make something draggable with mouse or finger; keeps it on screen.
local function draggable(object, handle, onDrop)
	local dragging, startInput, startPos, moved
	connect(handle.InputBegan, function(input)
		if isPress(input) then
			dragging, moved = true, false
			startInput = input.Position
			startPos = object.AbsolutePosition
			local done
			done = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					done:Disconnect()
					if moved and onDrop then
						onDrop()
					end
				end
			end)
		end
	end)
	connect(UserInputService.InputChanged, function(input)
		if not dragging or not isMove(input) then
			return
		end
		local delta = input.Position - startInput
		if not moved and delta.Magnitude < 4 then
			return
		end
		moved = true
		local parentSize = object.Parent.AbsoluteSize
		local size = object.AbsoluteSize
		local anchor = object.AnchorPoint
		local x = startPos.X + delta.X + anchor.X * size.X
		local y = startPos.Y + delta.Y + anchor.Y * size.Y
		x = math.clamp(x, anchor.X * size.X, math.max(anchor.X * size.X, parentSize.X - (1 - anchor.X) * size.X))
		y = math.clamp(y, anchor.Y * size.Y, math.max(anchor.Y * size.Y, parentSize.Y - (1 - anchor.Y) * size.Y))
		object.Position = UDim2.fromOffset(x, y)
	end)
	return function()
		return moved
	end
end

local function safeCall(fn, ...)
	if type(fn) ~= "function" then
		return
	end
	local ok, err = pcall(fn, ...)
	if not ok then
		warn("[Pepsi UI] callback error: " .. tostring(err))
	end
end

-- Keybind display names.
local function keyName(key)
	if key == nil then
		return "None"
	end
	if typeof(key) == "EnumItem" then
		local n = key.Name
		local short = {
			MouseButton1 = "MB1", MouseButton2 = "MB2", MouseButton3 = "MB3",
			LeftControl = "LCtrl", RightControl = "RCtrl", LeftShift = "LShift", RightShift = "RShift",
			LeftAlt = "LAlt", RightAlt = "RAlt", Return = "Enter", Backspace = "Bksp",
		}
		return short[n] or n
	end
	return tostring(key)
end

-- ======================================================================
-- The default logo: an accent tile with the first letter.
-- ======================================================================
local function defaultLogo(parent, size, letter)
	local tile = create("Frame", { Name = "Logo", BackgroundColor3 = Theme.AccentA, Size = UDim2.fromOffset(size, size), Parent = parent })
	corner(UDim.new(0.3, 0), tile)
	accentGradient(tile, 45)
	text({
		Parent = tile,
		Text = (letter or "P"):sub(1, 1):upper(),
		Font = FONT_BOLD,
		Size = math.floor(size * 0.55),
		Color = Color3.new(1, 1, 1),
		XAlign = Enum.TextXAlignment.Center,
		Sz = UDim2.fromScale(1, 1),
	})
	return tile
end

local function makeLogo(parent, size, logo, title)
	if type(logo) == "function" then
		local ok, made = pcall(logo, parent, size)
		if ok and made then
			return made
		end
	elseif logo ~= nil then
		local id = iconId(logo) or tostring(logo)
		local image = create("ImageLabel", { BackgroundTransparency = 1, Image = id, Size = UDim2.fromOffset(size, size), Parent = parent })
		corner(UDim.new(0.3, 0), image)
		return image
	end
	return defaultLogo(parent, size, title)
end

-- Size the window to the screen: big on phones, normal on PC.
function Window:FitToScreen()
	local viewport = workspace.CurrentCamera.ViewportSize
	if viewport.X < 10 then
		self.TargetScale = 1
		return
	end
	local fit = math.min(viewport.X * (self.Mobile and 0.94 or 0.9) / self.W, viewport.Y * (self.Mobile and 0.9 or 0.86) / self.H)
	local target = self.Mobile and math.clamp(fit, 0.5, 1.6) or math.clamp(fit, 0.5, 1)
	target *= (self.UserScale or 1)
	self.TargetScale = target
	if self.Scale and self.Open then
		tween(self.Scale, { Scale = target }, 0.3)
	end
end

function Window:SetScale(multiplier)
	self.UserScale = math.clamp(tonumber(multiplier) or 1, 0.5, 1.6)
	self:FitToScreen()
end

function Window:SetSidebar(collapsed, instant)
	self.Collapsed = collapsed
	local width = collapsed and SIDEBAR_COLLAPSED or SIDEBAR_W
	local t = instant and 0 or 0.35
	tween(self.Sidebar, { Size = UDim2.new(0, width, 1, 0) }, t)
	tween(self.Content, { Position = UDim2.fromOffset(width, 0), Size = UDim2.new(1, -width, 1, 0) }, t)
	for _, label in ipairs(self.CollapseLabels) do
		label.Visible = not collapsed
	end
	if self.SearchHit then
		self.SearchHit.Visible = collapsed
	end
	self.SearchInput.Parent.Size = collapsed and UDim2.fromOffset(40, 34) or UDim2.new(1, -24, 0, 34)
	for _, tab in ipairs(self.Tabs) do
		tab.Label.Visible = not collapsed
		tab.Button.Size = UDim2.new(1, 0, 0, 38)
	end
	local collapseIcon = self.CollapseButton and self.CollapseButton:FindFirstChild("Icon")
	if collapseIcon then
		collapseIcon.Image = iconId(collapsed and "panel-left-open" or "panel-left-close") or ""
	end
	self:RelayoutColumns()
end

function Window:SetOpen(state)
	self.Open = state
	if state then
		self.Holder.Visible = true
		tween(self.Scale, { Scale = self.TargetScale }, 0.4, Enum.EasingStyle.Back)
		tween(self.Main, { BackgroundTransparency = 0.03 }, 0.25)
	else
		tween(self.Scale, { Scale = self.TargetScale * 0.92 }, 0.2, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
		task.delay(0.18, function()
			if not self.Open then
				self.Holder.Visible = false
			end
		end)
	end
	if self.ToggleButton then
		tween(self.ToggleButton.Ring, { BackgroundTransparency = state and 0 or 0.4 }, 0.2)
	end
end

function Window:Toggle()
	self:SetOpen(not self.Open)
end

-- A small yes/no dialog inside the window.
function Window:Confirm(title, body, onYes)
	local modal = self.Modal
	for _, c in ipairs(modal:GetChildren()) do
		c:Destroy()
	end
	modal.Visible = true
	modal.BackgroundTransparency = 1
	tween(modal, { BackgroundTransparency = 0.45 }, 0.2)
	local card = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.55),
		Size = UDim2.fromOffset(320, 150),
		BackgroundColor3 = Theme.Section,
		ZIndex = 51,
		Parent = modal,
	})
	corner(14, card)
	stroke(card, 0.88)
	padding(card, 18, 18, 18, 18)
	text({ Parent = card, Text = title, Font = FONT_BOLD, Size = 17, Height = 20, Z = 52 })
	text({ Parent = card, Text = body, Color = Theme.SubText, Size = 13, Wrap = true, Pos = UDim2.fromOffset(0, 26), Sz = UDim2.new(1, 0, 0, 40), YAlign = Enum.TextYAlignment.Top, Z = 52 })
	local function close()
		tween(modal, { BackgroundTransparency = 1 }, 0.15)
		tween(card, { Position = UDim2.fromScale(0.5, 0.58) }, 0.15)
		task.delay(0.16, function()
			modal.Visible = false
			card:Destroy()
		end)
	end
	local function button(label, x, primary, fn)
		local b = create("TextButton", {
			AutoButtonColor = false,
			Text = label,
			FontFace = FONT_SEMI,
			TextSize = 14,
			TextColor3 = Theme.Text,
			BackgroundColor3 = primary and Theme.Error or Theme.Element,
			AnchorPoint = Vector2.new(0, 1),
			Position = UDim2.new(x, x > 0 and 5 or 0, 1, 0),
			Size = UDim2.new(0.5, -5, 0, 36),
			ZIndex = 52,
			Parent = card,
		})
		corner(10, b)
		b.Activated:Connect(function()
			close()
			if fn then
				safeCall(fn)
			end
		end)
	end
	button("Cancel", 0, false)
	button("Yes", 0.5, true, onYes)
	tween(card, { Position = UDim2.fromScale(0.5, 0.5) }, 0.3, Enum.EasingStyle.Back)

	local gui = newScreen("PepsiUI_Button", 210)

	})

		logo.AnchorPoint = Vector2.new(0.5, 0.5)
		logo.Position = UDim2.fromScale(0.5, 0.5)
		for _, d in ipairs(logo:GetDescendants()) do
			if d:IsA("GuiObject") then
				d.ZIndex = d.ZIndex + 2
			end
		end
		logo.ZIndex = 3
	end
	local scaler = create("UIScale", { Parent = button })
	local wasDragged = draggable(button, button)
	button.MouseEnter:Connect(function()
		tween(scaler, { Scale = 1.07 }, 0.15)
	end)
	button.MouseLeave:Connect(function()
		tween(scaler, { Scale = 1 }, 0.15)

	if opts.ShowButton == false or (opts.ShowButton == nil and not self.Mobile and opts.PCButton == false) then
		button.Visible = false
	end

-- Top bar: name · game · fps · ping · time. Drag it anywhere.
function Window:Watermark(customText)
	if self.WatermarkFrame then
		self.WatermarkFrame.Visible = true
		return self.WatermarkFrame
	end
	local bar = create("Frame", {

	local ping = seg("0 ms", 4, Theme.SubText, FONT_MONO)
	local clock = seg("00:00", 5, Theme.SubText, FONT_MONO)
	draggable(bar, bar)
	local frames, last = 0, os.clock()
	connect(RunService.RenderStepped, function()
		frames += 1
		local now = os.clock()
		if now - last >= 1 then
			fps.Text = frames .. " fps"
			frames, last = 0, now
			pcall(function()
				ping.Text = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) .. " ms"
			end)
			clock.Text = os.date("%H:%M")
		end
	end)

-- Search: hide what doesn't match on the open page.
function Window:Filter(query)
	query = tostring(query or ""):lower()
	local page = self.CurrentPage
	if not page then
		return
	end
	for _, section in ipairs(page.Sections) do
		local any = false
		for _, element in ipairs(section.Elements) do
			local match = query == "" or element.SearchText:find(query, 1, true) ~= nil or section.Name:lower():find(query, 1, true) ~= nil
			element.Frame.Visible = match and not element.Hidden
			any = any or match
		end
		section.Frame.Visible = any or query == ""
	end
end

-- Two columns when there's room, one on narrow windows.
function Window:RelayoutColumns()
	local width = self.W - (self.Collapsed and SIDEBAR_COLLAPSED or SIDEBAR_W)
	local two = width >= 520
	for _, tab in ipairs(self.Tabs) do
		for _, page in ipairs(tab.Pages) do
			page.Left.Size = two and UDim2.new(0.5, -6, 0, 0) or UDim2.new(1, 0, 0, 0)
			page.Right.Visible = two
			for _, section in ipairs(page.Sections) do
				section.Frame.Parent = (two and section.Side == "Right") and page.Right or page.Left
			end
		end
	end
end

function Window:SelectTab(tab, pageIndex)
	local previous = self.CurrentTab
	if previous and previous ~= tab then
		tween(previous.Button, { BackgroundTransparency = 1 }, 0.2)
		tween(previous.Label, { TextColor3 = Theme.SubText }, 0.2)
		tween(previous.Bar, { Size = UDim2.fromOffset(3, 0) }, 0.2)
		previous.Icon.ImageColor3 = Theme.SubText
		previous.IconGradient.Enabled = false
		if previous.Letter then
			previous.Letter.TextColor3 = Theme.SubText
		end
	end
	self.CurrentTab = tab
	tween(tab.Button, { BackgroundTransparency = 0.25 }, 0.25)
	tween(tab.Label, { TextColor3 = Theme.Text }, 0.25)
	tween(tab.Bar, { Size = UDim2.fromOffset(3, 18) }, 0.35, Enum.EasingStyle.Back)
	tab.Icon.ImageColor3 = Color3.new(1, 1, 1)
	tab.IconGradient.Enabled = true
	if tab.Letter then
		tab.Letter.TextColor3 = Theme.AccentA
	end
	self.PageTitle.Text = tab.Name
	self.PageDesc.Text = tab.Description ~= "" and tab.Description or (self.Subtitle ~= "" and self.Subtitle or "")
	if #tab.Pages == 0 then
		newPage(self, tab, tab.Name)
	end
	-- Sub tab bar.
	for _, c in ipairs(self.SubPills:GetChildren()) do
		if c:IsA("GuiButton") then
			c:Destroy()
		end
	end
	local showBar = #tab.Pages > 1
	self.SubBar.Visible = showBar
	self.Pages.Position = UDim2.fromOffset(0, showBar and 104 or 62)
	self.Pages.Size = UDim2.new(1, 0, 1, showBar and -104 or -62)
	if showBar then
		for i, page in ipairs(tab.Pages) do
			local pill = create("TextButton", {
				Name = "Pill" .. i,
				AutoButtonColor = false,
				Text = page.Name,
				FontFace = FONT_SEMI,
				TextSize = 13,
				TextColor3 = Theme.SubText,
				BackgroundTransparency = 1,
				Size = UDim2.fromOffset(0, 28),
				AutomaticSize = Enum.AutomaticSize.X,
				LayoutOrder = i,
				ZIndex = 5,
				Parent = self.SubPills,
			})
			padding(pill, 0, 14, 0, 14)
			page.Pill = pill
			pill.Activated:Connect(function()
				self:SelectPage(tab, i)
			end)
		end
	end
	self:SelectPage(tab, pageIndex or tab.LastPage or 1)
end

function Window:SelectPage(tab, index)
	local page = tab.Pages[index] or tab.Pages[1]
	tab.LastPage = index
	for _, t in ipairs(self.Tabs) do
		for _, p in ipairs(t.Pages) do
			if p ~= page then
				p.Scroll.Visible = false
			end
		end
	end
	self.CurrentPage = page
	page.Scroll.Visible = true
	page.Scroll.Position = UDim2.fromOffset(0, 18)
	tween(page.Scroll, { Position = UDim2.fromOffset(0, 0) }, 0.4)
	for _, p in ipairs(tab.Pages) do
		if p.Pill then
			tween(p.Pill, { TextColor3 = p == page and Theme.Text or Theme.SubText }, 0.2)
		end
	end
	if page.Pill then
		task.defer(function()
			local bar = self.SubBar
			local ratio = bar.AbsoluteSize.Y / 36
			if ratio <= 0 then
				return
			end
			local rel = (page.Pill.AbsolutePosition - bar.AbsolutePosition) / ratio
			tween(self.SubHighlight, {
				Position = UDim2.fromOffset(rel.X, 4),
				Size = UDim2.fromOffset(page.Pill.AbsoluteSize.X / ratio, 28),
			}, 0.35, Enum.EasingStyle.Quint)
		end)
	end
	if self.SearchInput.Text ~= "" then
		self:Filter(self.SearchInput.Text)
	end
end

function Tab:SubTab(opts)
	if type(opts) == "string" then
		opts = { Name = opts }
	end
	local page = newPage(self.Window, self, (opts and opts.Name) or ("Page " .. (#self.Pages + 1)))
	local sub = setmetatable({ Page = page, Window = self.Window, Tab = self }, Tab)
	self.Window:RelayoutColumns()
	if self.Window.CurrentTab == self then
		self.Window:SelectTab(self, self.LastPage)
	end
	return sub
end

	self.Window:SelectTab(self.Tab or self)
end

local function register(flag, element, kind)
	element.Type = kind
	element.Flag = flag
	if flag then
		Library.Options[flag] = element
		Library.Flags[flag] = element.Value
	end
end

local function makeElement(section, name, description, height)
	local window = section.Window
	section.Count += 1
	local base = window.RowHeight
	local h = height or (description and base + 16 or base)
	local frame = create("Frame", {
		Name = "Element_" .. tostring(name),
		Size = UDim2.new(1, 0, 0, h),
		BackgroundColor3 = Theme.Element,
		BackgroundTransparency = 0.4,
		LayoutOrder = section.Count,
		ClipsDescendants = true,
		ZIndex = 3,
		Parent = section.Frame,
	})
	corner(10, frame)
	local s = stroke(frame, 0.95)
	local title = text({
		Parent = frame,
		Text = tostring(name or ""),
		Font = FONT_SEMI,
		Size = 14,
		Pos = UDim2.fromOffset(12, description and 8 or (base - 18) / 2),
		Sz = UDim2.new(1, -24, 0, 18),
		Truncate = true,
		Z = 4,
	})
	local desc
	if description then
		desc = text({ Parent = frame, Text = description, Color = Theme.SubText, Size = 12, Pos = UDim2.fromOffset(12, 27), Sz = UDim2.new(1, -24, 0, 15), Truncate = true, Z = 4 })
	end
	local element = {
		Frame = frame,
		Title = title,
		Desc = desc,
		Stroke = s,
		Section = section,
		SearchText = (tostring(name or "") .. " " .. tostring(description or "")):lower(),
		Height = h,
		Listeners = {},
	}
	function element:OnChanged(fn)
		table.insert(self.Listeners, fn)
	end
	function element:SetVisible(visible)
		self.Hidden = not visible
		self.Frame.Visible = visible
	end
	function element:SetTitle(t)
		self.Title.Text = tostring(t)
		self.SearchText = tostring(t):lower()
	end
	function element:SetDescription(t)
		if self.Desc then
			self.Desc.Text = tostring(t)
		end
	end
	function element:Fire(...)
		for _, fn in ipairs(self.Listeners) do
			safeCall(fn, ...)
		end
	end
	function element:Hover(on)
		tween(frame, { BackgroundTransparency = on and 0.15 or 0.4 }, 0.18)
		tween(s, { Transparency = on and 0.9 or 0.95 }, 0.18)
	end
	frame.MouseEnter:Connect(function()
		element:Hover(true)
	end)
	frame.MouseLeave:Connect(function()
		element:Hover(false)
	end)
	table.insert(section.Elements, element)
	return element
end

	end)

	if e.Value then
		task.defer(safeCall, opts.Callback, true)
	end
	return e

-- ---------------- Slider ----------------
function Section:Slider(opts)
	opts = opts or {}
	local min, max = tonumber(opts.Min) or 0, tonumber(opts.Max) or 100
	local step = tonumber(opts.Increment or opts.Step or opts.Float) or 1
	local e = makeElement(self, opts.Name, opts.Description, (opts.Description and self.Window.RowHeight + 16 or self.Window.RowHeight) + 18)
	e.Title.Size = UDim2.new(1, -110, 0, 18)
	local function snap(v)
		v = math.clamp(tonumber(v) or min, min, max)
		v = min + math.floor((v - min) / step + 0.5) * step
		local decimals = math.max(0, math.ceil(-math.log10(step) - 1e-9))
		local m = 10 ^ decimals
		return math.clamp(math.floor(v * m + 0.5) / m, min, max)
	end
	e.Value = snap(opts.Default or min)
	local valueBox = create("TextBox", {
		Name = "Value",
		AnchorPoint = Vector2.new(1, 0),
		Position = UDim2.new(1, -12, 0, e.Desc and 8 or (self.Window.RowHeight - 24) / 2),
		Size = UDim2.fromOffset(0, 22),
		AutomaticSize = Enum.AutomaticSize.X,
		BackgroundColor3 = Theme.Field,
		FontFace = FONT_MONO,
		TextSize = 12,
		TextColor3 = Theme.Text,
		Text = "",
		ClearTextOnFocus = false,
		ZIndex = 6,
		Parent = e.Frame,
	})
	corner(7, valueBox)
	stroke(valueBox, 0.9)
	padding(valueBox, 0, 8, 0, 8)
	local track = create("Frame", {
		Name = "Track",
		AnchorPoint = Vector2.new(0, 1),
		Position = UDim2.new(0, 12, 1, -14),
		Size = UDim2.new(1, -24, 0, 6),
		BackgroundColor3 = Theme.Field,
		ZIndex = 4,
		Parent = e.Frame,
	})
	corner(UDim.new(1, 0), track)
	stroke(track, 0.92)
	local fill = create("Frame", { Name = "Fill", Size = UDim2.fromScale(0, 1), BackgroundColor3 = Theme.AccentA, ZIndex = 5, Parent = track })
	corner(UDim.new(1, 0), fill)
	accentGradient(fill)
	local knob = create("Frame", {
		Name = "Knob",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0, 0.5),
		Size = UDim2.fromOffset(14, 14),
		BackgroundColor3 = Color3.new(1, 1, 1),
		ZIndex = 6,
		Parent = track,
	})
	corner(UDim.new(1, 0), knob)
	local knobStroke = stroke(knob, 0, 2, Theme.AccentA)
	accentSolid(knobStroke, "Color")
	local suffix = opts.Suffix or ""
	local function render(animate)
		local alpha = max == min and 0 or (e.Value - min) / (max - min)
		local t = animate and 0.12 or 0
		tween(fill, { Size = UDim2.fromScale(alpha, 1) }, t)
		tween(knob, { Position = UDim2.fromScale(alpha, 0.5) }, t)
		valueBox.Text = tostring(e.Value) .. suffix
	end
	function e:Set(value)
		local v = snap(value)
		local changed = v ~= self.Value
		self.Value = v
		if self.Flag then
			Library.Flags[self.Flag] = v
		end
		render(true)
		if changed then
			safeCall(opts.Callback, v)
			self:Fire(v)
		end
	end
	function e:Get()
		return self.Value
	end
	local dragging = false
	local function fromInput(input)
		local x = (input.Position.X - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X)
		e:Set(min + math.clamp(x, 0, 1) * (max - min))
	end
	local hit = create("TextButton", {
		Name = "Hit",
		AutoButtonColor = false,
		BackgroundTransparency = 1,
		Text = "",
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, -6, 0.5, 0),
		Size = UDim2.new(1, 12, 0, 28),
		ZIndex = 7,
		Parent = track,
	})
	hit.InputBegan:Connect(function(input)
		if isPress(input) then
			dragging = true
			e.Section.Page.Scroll.ScrollingEnabled = false
			tween(knob, { Size = UDim2.fromOffset(18, 18) }, 0.15)
			fromInput(input)
		end
	end)
	connect(UserInputService.InputChanged, function(input)
		if dragging and isMove(input) then
			fromInput(input)
		end
	end)
	connect(UserInputService.InputEnded, function(input)
		if dragging and isPress(input) then
			dragging = false
			e.Section.Page.Scroll.ScrollingEnabled = true
			tween(knob, { Size = UDim2.fromOffset(14, 14) }, 0.2, Enum.EasingStyle.Back)
		end
	end)
	valueBox.FocusLost:Connect(function()
		local n = tonumber((valueBox.Text:gsub("[^%d%.%-]", "")))
		if n then
			e:Set(n)
		end
		render(false)
	end)
	register(opts.Flag, e, "Slider")
	render(false)
	return e
end

-- ---------------- Dropdown ----------------
function Section:Dropdown(opts)
	opts = opts or {}
	local window = self.Window
	local e = makeElement(self, opts.Name, opts.Description)
	local base = e.Height
	e.Title.Size = UDim2.new(0.5, -12, 0, 18)
	e.Multi = opts.Multi == true or opts.MultiSelect == true
	e.Options = {}
	for _, v in ipairs(opts.Options or opts.Values or {}) do
		table.insert(e.Options, tostring(v))
	end
	if e.Multi then
		e.Value = {}
		local d = opts.Default
		if type(d) == "table" then
			for k, v in pairs(d) do
				if v == true then
					table.insert(e.Value, tostring(k))
				elseif type(k) == "number" then
					table.insert(e.Value, tostring(v))
				end
			end
		elseif d ~= nil then
			e.Value = { tostring(d) }
		end
	else
		local d = opts.Default
		if type(d) == "number" and e.Options[d] then
			d = e.Options[d]
		end
		e.Value = d ~= nil and tostring(d) or nil
	end

	local valueLabel = text({
		Parent = e.Frame,
		Text = "",
		Color = Theme.SubText,
		Size = 13,
		XAlign = Enum.TextXAlignment.Right,
		Anchor = Vector2.new(1, 0),
		Pos = UDim2.new(1, -36, 0, (base - 18) / 2),
		Sz = UDim2.new(0.5, -30, 0, 18),
		Truncate = true,
		Z = 4,
	})
	local chevron = icon(e.Frame, "chevron-down", 16, Theme.SubText, { AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -12, 0, (base - 16) / 2), ZIndex = 4 })
	local head = create("TextButton", { Name = "Hit", AutoButtonColor = false, BackgroundTransparency = 1, Text = "", Size = UDim2.new(1, 0, 0, base), ZIndex = 6, Parent = e.Frame })

	local panel = create("Frame", {
		Name = "Panel",
		Position = UDim2.fromOffset(8, base),
		Size = UDim2.new(1, -16, 0, 0),
		BackgroundTransparency = 1,
		ZIndex = 4,
		Parent = e.Frame,
	})
	local search = create("TextBox", {
		Name = "Search",
		Size = UDim2.new(1, 0, 0, 28),
		BackgroundColor3 = Theme.Field,
		FontFace = FONT,
		TextSize = 13,
		TextColor3 = Theme.Text,
		PlaceholderText = "Search...",
		PlaceholderColor3 = Theme.Dim,
		Text = "",
		ClearTextOnFocus = false,
		TextXAlignment = Enum.TextXAlignment.Left,
		ZIndex = 5,
		Parent = panel,
	})
	corner(8, search)
	padding(search, 0, 10, 0, 10)
	local optionList = create("ScrollingFrame", {
		Name = "List",
		Position = UDim2.fromOffset(0, 34),
		Size = UDim2.new(1, 0, 0, 0),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 2,
		ScrollBarImageColor3 = Theme.AccentA,
		CanvasSize = UDim2.new(),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ZIndex = 5,
		Parent = panel,
	})
	list(optionList, 3)
	padding(optionList, 0, 4, 6, 0)

	local function isSelected(v)
		if e.Multi then
			return table.find(e.Value, v) ~= nil
		end
		return e.Value == v
	end
	local function displayText()
		if e.Multi then
			return #e.Value == 0 and "None" or table.concat(e.Value, ", ")
		end
		return e.Value or "None"
	end
	local optionButtons = {}
	local open = false
	local function panelHeight()
		local visible = 0
		for _, b in pairs(optionButtons) do
			if b.Visible then
				visible += 1
			end
		end
		local shown = math.min(visible, 6)
		return 34 + shown * 31 + 8
	end
	local function resize(animate)
		local h = open and (base + panelHeight()) or base
		optionList.Size = UDim2.new(1, 0, 0, open and (panelHeight() - 40) or 0)
		tween(e.Frame, { Size = UDim2.new(1, 0, 0, h) }, animate and 0.3 or 0)
		tween(chevron, { Rotation = open and 180 or 0 }, 0.3)
	end
	local function paint()
		valueLabel.Text = displayText()
		for v, b in pairs(optionButtons) do
			local on = isSelected(v)
			tween(b, { BackgroundTransparency = on and 0.1 or 1 }, 0.15)
			tween(b.Label, { TextColor3 = on and Theme.Text or Theme.SubText }, 0.15)
			b.Check.Visible = on
		end
	end
	local function fire()
		if e.Flag then
			Library.Flags[e.Flag] = e.Value
		end
		local value = e.Value
		if e.Multi then
			value = table.clone(e.Value)
		end
		safeCall(opts.Callback, value)
		e:Fire(value)
	end
	local function choose(v)
		if e.Multi then
			local i = table.find(e.Value, v)
			if i then
				table.remove(e.Value, i)
			else
				table.insert(e.Value, v)
			end
		else
			if e.Value == v and opts.AllowNull then
				e.Value = nil
			else
				e.Value = v
			end
			open = false
			resize(true)
		end
		paint()
		fire()
	end
	local function build()
		for _, b in pairs(optionButtons) do
			b:Destroy()
		end
		optionButtons = {}
		for i, v in ipairs(e.Options) do
			local b = create("TextButton", {
				Name = "Option",
				AutoButtonColor = false,
				Text = "",
				Size = UDim2.new(1, 0, 0, 28),
				BackgroundColor3 = Theme.ElementHover,
				BackgroundTransparency = 1,
				LayoutOrder = i,
				ZIndex = 6,
				Parent = optionList,
			})
			corner(7, b)
			local label = text({ Parent = b, Name = "Label", Text = v, Size = 13, Color = Theme.SubText, Pos = UDim2.fromOffset(10, 5), Sz = UDim2.new(1, -36, 0, 18), Truncate = true, Z = 7 })
			local check = icon(b, "check", 14, Theme.AccentA, { Name = "Check", AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -8, 0.5, 0), ZIndex = 7 })
			accentSolid(check, "ImageColor3")
			b.Label, b.Check = label, check
			b.MouseEnter:Connect(function()
				if not isSelected(v) then
					tween(b, { BackgroundTransparency = 0.5 }, 0.12)
				end
			end)
			b.MouseLeave:Connect(function()
				if not isSelected(v) then
					tween(b, { BackgroundTransparency = 1 }, 0.12)
				end
			end)
			b.Activated:Connect(function()
				choose(v)
			end)
			optionButtons[v] = b
		end
		paint()
	end
	head.Activated:Connect(function()
		open = not open
		search.Text = ""
		resize(true)
	end)
	search:GetPropertyChangedSignal("Text"):Connect(function()
		local q = search.Text:lower()
		for v, b in pairs(optionButtons) do
			b.Visible = q == "" or v:lower():find(q, 1, true) ~= nil
		end
		resize(false)
	end)
	function e:Set(value)
		if e.Multi then
			local new = {}
			if type(value) == "table" then
				for k, v in pairs(value) do
					if v == true then
						table.insert(new, tostring(k))
					elseif type(k) == "number" then
						table.insert(new, tostring(v))
					end
				end
			elseif value ~= nil then
				new = { tostring(value) }
			end
			e.Value = new
		else
			if type(value) == "table" then
				value = value[1]
			end
			e.Value = value ~= nil and tostring(value) or nil
		end
		paint()
		fire()
	end
	function e:Get()
		return e.Value
	end
	function e:Refresh(options, keep)
		e.Options = {}
		for _, v in ipairs(options or {}) do
			table.insert(e.Options, tostring(v))
		end
		if not keep then
			if e.Multi then
				local kept = {}
				for _, v in ipairs(e.Value) do
					if table.find(e.Options, v) then
						table.insert(kept, v)
					end
				end
				e.Value = kept
			elseif e.Value and not table.find(e.Options, e.Value) then
				e.Value = nil
			end
		end
		build()
		resize(false)
	end
	e.SetValues = e.Refresh
	build()
	register(opts.Flag, e, "Dropdown")
	return e
end

-- ---------------- Keybind ----------------
Library.Keybinds = {}
function Section:Keybind(opts)
	opts = opts or {}
	local e = makeElement(self, opts.Name, opts.Description)
	e.Title.Size = UDim2.new(1, -110, 0, 18)
	local default = opts.Default
	if type(default) == "string" then
		default = Enum.KeyCode[default] or Enum.UserInputType[default]
	end
	e.Value = default
	e.Mode = opts.Mode or "Toggle"
	e.State = false
	local pill = create("TextButton", {
		AutoButtonColor = false,
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -12, 0.5, 0),
		Size = UDim2.fromOffset(0, 26),
		AutomaticSize = Enum.AutomaticSize.X,
		BackgroundColor3 = Theme.Field,
		FontFace = FONT_MONO,
		TextSize = 12,
		TextColor3 = Theme.Text,
		Text = keyName(e.Value),
		ZIndex = 6,
		Parent = e.Frame,
	})
	corner(7, pill)
	local pillStroke = stroke(pill, 0.88)
	padding(pill, 0, 10, 0, 10)
	local listening = false
	pill.Activated:Connect(function()
		listening = true
		pill.Text = "..."
		tween(pillStroke, { Color = Theme.AccentA, Transparency = 0.2 }, 0.15)
	end)
	function e:Set(key)
		if type(key) == "string" then
			key = Enum.KeyCode[key] or Enum.UserInputType[key]
		end
		self.Value = key
		pill.Text = keyName(key)
		if self.Flag then
			Library.Flags[self.Flag] = key
		end
		safeCall(opts.ChangedCallback, key)
	end
	function e:Get()
		return self.Value
	end
	connect(UserInputService.InputBegan, function(input, processed)
		if listening then
			if input.UserInputType == Enum.UserInputType.Keyboard then
				listening = false
				tween(pillStroke, { Color = Theme.Stroke, Transparency = 0.88 }, 0.15)
				if input.KeyCode == Enum.KeyCode.Escape or input.KeyCode == Enum.KeyCode.Backspace then
					e:Set(nil)
				else
					e:Set(input.KeyCode)
				end
			elseif input.UserInputType == Enum.UserInputType.MouseButton2 or input.UserInputType == Enum.UserInputType.MouseButton3 then
				listening = false
				tween(pillStroke, { Color = Theme.Stroke, Transparency = 0.88 }, 0.15)
				e:Set(input.UserInputType)
			end
			return
		end
		if processed or e.Value == nil or Library.Unloaded then
			return
		end
		if input.KeyCode == e.Value or input.UserInputType == e.Value then
			if e.Mode == "Hold" then
				e.State = true
			else
				e.State = not e.State
			end
			safeCall(opts.Callback, e.State)
			e:Fire(e.State)
		end
	end)
	connect(UserInputService.InputEnded, function(input)
		if e.Mode == "Hold" and e.Value and (input.KeyCode == e.Value or input.UserInputType == e.Value) then
			e.State = false
			safeCall(opts.Callback, false)
			e:Fire(false)
		end
	end)
	register(opts.Flag, e, "Keybind")
	table.insert(Library.Keybinds, e)
	return e
end

	local function setValue(v)
		v = math.clamp(v, min, max)
		local changed = v ~= e.Value
		e.Value = v
		valueLabel.Text = tostring(v)
		if e.Flag then
			Library.Flags[e.Flag] = v
		end

			safeCall(opts.Callback, v)
			e:Fire(v)
		end
	end

		local b = create("TextButton", {
			AutoButtonColor = false,
			Text = "",
			AnchorPoint = Vector2.new(i == 1 and 0 or 1, 0),
			Position = UDim2.fromScale(i == 1 and 0 or 1, 0),
			Size = UDim2.fromOffset(28, 28),
			BackgroundTransparency = 1,
			ZIndex = 6,
			Parent = holder,
		})
		local img = icon(b, glyph, 14, Theme.SubText, { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), ZIndex = 6 })
		b.Activated:Connect(function()
			setValue(e.Value + (i == 1 and -step or step))

		setValue(tonumber(v) or e.Value)

-- ======================================================================

local function canFiles()
	return writefile and readfile and isfile and isfolder and makefolder
end

local function ensureFolder(path)
	if not canFiles() then
		return
	end
	local built = ""
	for part in path:gmatch("[^/]+") do
		built = built == "" and part or (built .. "/" .. part)
		if not isfolder(built) then
			pcall(makefolder, built)
		end
	end
end

local function encodeValue(element)
	local v = element.Value
	if element.Type == "Colorpicker" and typeof(v) == "Color3" then
		return { t = "color", v = v:ToHex() }
	elseif element.Type == "Keybind" then
		return { t = "key", v = v and v.Name or nil, m = element.Mode }
	elseif type(v) == "table" then
		return { t = "list", v = v }
	end
	return { t = "value", v = v }
end

local function decodeValue(element, data)
	if type(data) ~= "table" then
		return
	end
	if data.t == "color" then
		local ok, c = pcall(Color3.fromHex, data.v)
		if ok then
			element:Set(c)
		end
	elseif data.t == "key" then
		element:Set(data.v)
	else
		element:Set(data.v)
	end
end

function Window:ConfigList()
	local names = {}
	if not canFiles() or not listfiles then
		return names
	end
	ensureFolder(self.ConfigFolder)
	for _, path in ipairs(listfiles(self.ConfigFolder)) do
		local name = tostring(path):match("([^/\\]+)%.json$")
		if name then
			table.insert(names, name)
		end
	end
	table.sort(names)
	return names
end

function Window:SaveConfig(name)
	name = tostring(name or ""):gsub("[^%w%s%-_]", "")
	if name == "" or not canFiles() then
		return false
	end
	ensureFolder(self.ConfigFolder)
	local data = {}
	for flag, element in pairs(Library.Options) do
		if element.Window == nil or element.Window == self then
			if not element.NoSave then
				data[flag] = encodeValue(element)
			end
		end
	end
	return pcall(writefile, self.ConfigFolder .. "/" .. name .. ".json", HttpService:JSONEncode(data))
end

function Window:LoadConfig(name)
	local path = self.ConfigFolder .. "/" .. tostring(name) .. ".json"
	if not canFiles() or not isfile(path) then
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(path))
	end)
	if not ok or type(data) ~= "table" then
		return false
	end
	for flag, value in pairs(data) do
		local element = Library.Options[flag]
		if element and not element.NoSave then
			pcall(decodeValue, element, value)
		end
	end
	return true
end

function Window:DeleteConfig(name)
	local path = self.ConfigFolder .. "/" .. tostring(name) .. ".json"
	if canFiles() and delfile and isfile(path) then
		pcall(delfile, path)
		return true
	end
	return false
end

function Window:SetAutoload(name)
	if not canFiles() then
		return
	end
	ensureFolder(self.ConfigFolder)
	if name then
		pcall(writefile, self.ConfigFolder .. "/autoload.txt", tostring(name))
	elseif delfile and isfile(self.ConfigFolder .. "/autoload.txt") then
		pcall(delfile, self.ConfigFolder .. "/autoload.txt")
	end
end

function Window:GetAutoload()
	if not canFiles() or not isfile(self.ConfigFolder .. "/autoload.txt") then
		return nil
	end
	local ok, name = pcall(readfile, self.ConfigFolder .. "/autoload.txt")
	return ok and name ~= "" and name or nil
end

-- Loads the autoload config, if you picked one. Call after building your tabs.
function Window:LoadAutoload()
	local name = self:GetAutoload()
	if name and self:LoadConfig(name) then
		Library:Notify({ Title = "Config loaded", Content = "Autoloaded \"" .. name .. "\".", Type = "Success", Duration = 3 })
	end
end

-- ======================================================================
-- Unload
-- ======================================================================
function Library:OnUnload(fn)
	table.insert(self.UnloadHooks, fn)
end

function Library:Unload()
	if self.Unloaded then
		return
	end
	self.Unloaded = true
	for _, fn in ipairs(self.UnloadHooks) do
		safeCall(fn)
	end
	for _, c in ipairs(self.Connections) do
		pcall(function()
			c:Disconnect()
		end)
	end
	for _, window in ipairs(self.Windows) do
		pcall(function()
			window.Gui:Destroy()
		end)
		pcall(function()
			window.ButtonGui:Destroy()
		end)
	end
	pcall(function()
		if notifyGui then
			notifyGui:Destroy()
		end
	end)
end

return Library

task.spawn(function()
	local n = 0
	while not UI.Unloaded do
		n += 1
		status:SetContent("Live update #" .. n .. " — paragraphs can show status text that changes.")
		task.wait(1)
	end
end)
UI:Notify({ Title = "Pepsi Hub", Content = UI.Mobile and "Tap the logo to open and close the menu." or "Press Right Ctrl or click the logo to open and close the menu.", Duration = 6 })
end, function(err)
	return tostring(err) .. "\n" .. debug.traceback()
end)
if PEPSI_DEMO_OK then
	print("[Pepsi UI] demo loaded")
else
	warn("[Pepsi UI] demo failed: " .. tostring(PEPSI_DEMO_ERR))
	pepsiToast("Pepsi UI error", tostring(PEPSI_DEMO_ERR):sub(1, 180), 20)
end
